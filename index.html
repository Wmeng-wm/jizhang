function aiParseText(text) { var result = { amount: 0, category: null, note: '', type: 'expense' }; if (!text) return result; text = String(text); // ===== 1. 金额识别（增强版）===== var candidates = []; // 匹配所有带两位小数金额 var matches = text.match(/[\d,]+\.\d{2}/g); if (matches) { matches.forEach(function(v) { var num = parseFloat(v.replace(/,/g, '')); if (!isNaN(num)) candidates.push(num); }); } // 匹配 ¥ / ￥ var yuanMatches = text.match(/[¥￥]\s*([\d,]+\.?\d*)/g); if (yuanMatches) { yuanMatches.forEach(function(v) { var num = parseFloat(v.replace(/[¥￥,\s]/g, '')); if (!isNaN(num)) candidates.push(num); }); } if (candidates.length > 0) { // 取最合理金额（通常支付金额不是最大的余额） candidates.sort((a, b) => b - a); // 排除明显余额（太大） var filtered = candidates.filter(function(n) { return n < 50000; }); result.amount = filtered.length ? filtered[0] : candidates[0]; } // ===== 2. 收入识别 ===== if (/工资|收入|报销|退款|转入|收款|到账/.test(text)) { result.type = 'income'; } // ===== 3. 分类识别（增强版）===== var catRules = [ { cat: '餐饮', keywords: [ '美团','饿了么','外卖','餐厅','咖啡','奶茶', '早餐','午餐','晚餐','烧烤','火锅','吃饭' ] }, { cat: '飞机', keywords: [ '机票','航班','航空','南航','东航','国航','登机' ] }, { cat: '高铁', keywords: [ '12306','高铁','动车','火车票','铁路' ] }, { cat: '网约车', keywords: [ '滴滴','曹操','T3','打车','出租车','网约车' ] }, { cat: '酒店', keywords: [ '酒店','宾馆','汉庭','全季','亚朵','如家','维也纳' ] }, { cat: '购物', keywords: [ '淘宝','京东','拼多多','购物' ] }, { cat: '超市', keywords: [ '超市','沃尔玛','永辉','便利店' ] }, { cat: '通讯', keywords: [ '话费','流量','宽带','移动','联通','电信' ] }, { cat: '医疗', keywords: [ '医院','药店','诊所','药' ] } ]; var bestCat = null; var bestScore = 0; catRules.forEach(function(rule) { var score = 0; rule.keywords.forEach(function(word) { if (text.indexOf(word) >= 0) score++; }); if (score > bestScore) { bestScore = score; bestCat = rule.cat; } }); if (bestCat) { result.category = bestCat; } else { result.category = result.type === 'income' ? getCategories('income')[0].name : getCategories('expense')[0].name; } // ===== 4. 备注提取 ===== var merchant = text.match(/([一-龥A-Za-z]{2,20}(?:酒店|宾馆|餐厅|店))/); if (merchant) { result.note = merchant[1]; } else { result.note = text .replace(/[¥￥\d,\.\s]/g, '') .substring(0, 20); } return result; }
